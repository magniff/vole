// VARIANTS: #[pure] -
// (0.30.1) An extern's declared effects changed between two builds with
// one cache: with #[pure], -O2 may drop or merge the calls whose results
// are unused or repeated, without it not. The declaration is hashed by its
// effects, not only by whether there is one.
extern "C" {
    @V@
    fn ext(x: i32) -> i32;
}

#[inline(never)]
fn user(x: i32) -> i32 {
    // UNSAFE: C calls. SAFETY: ext is a C function of an int.
    unsafe {
        ext(x);
        ext(x) + ext(x)
    }
}

fn main() {
    println("{}", user(3));
}
