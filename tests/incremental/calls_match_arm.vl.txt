// VARIANTS: 5 4
// (0.30.1, RT-CG-3) tests/incremental.sh builds this with one cache as
// each variant after the other: 5 (the arm holding the program's only call
// of Vec::insert is never taken), then 4 (it is), and the other way round.
// 0.30.0's warm build kept an out-of-line Vec::insert the cold build does
// not have.
fn parse(s: &str) -> i64 {
    let mut n: i64 = 0;
    let mut neg = false;
    for b in s.bytes() {
        if b == b'-' { neg = true; } else { n = n * 10 + (b - b'0') as i64; }
    }
    if neg { -n } else { n }
}

fn main(gpa: Gpa) {
    let o0: i8 = parse("-71") as i8;
    let o2: i64 = parse("-21");
    let mut v: Vec<i16, Gpa> = Vec::new_in(gpa);
    let x: u8 = ((o0 as u8) & 15u8) * 7u8;
    match 4i8 {
        @V@i8..=9i8 => {
            if o2 % 3 == 0 {
                let at: usize = (x as usize) % (v.len() + 1);
                v.insert(at, v.len() as i16);
            }
        }
        _ => {}
    }
    println("{} {}", v.len(), x);
}
