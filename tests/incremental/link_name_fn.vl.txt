// VARIANTS: fa fb
// (0.30.1, RT-TL-2) An extern's #[link_name] changed between two builds
// with one cache: user's code calls the new symbol. 0.30.0 hashed only
// whether the extern had a link name, and a warm -O1 or -O2 build called
// the old one. (A C function is named by the symbol it links to since
// 0.30.1, so the name in the hash is the link name.)
extern "C" {
    #[link_name = "@V@"]
    fn ext(x: i32) -> i32;
}

#[inline(never)]
fn user(x: i32) -> i32 {
    // UNSAFE: a C call. SAFETY: ext is a C function of an int.
    unsafe { ext(x) }
}

fn main() {
    println("{}", user(3));
}
