// VARIANTS: 0 1
// (0.30.1, RT-CG-3) tests/incremental.sh builds this with one cache as
// each variant after the other: 0 (an endless loop: the loop over v is
// never reached, and its calls are gone), then 1, and the other way round.
fn main(gpa: Gpa) {
    let mut h: u64 = 1;
    let v: Vec<i64, Gpa> = (1i64..9i64).collect_in(gpa);
    let mut w: u32 = 0;
    loop {
        w += @V@;
        if w > 1 { break; }
    }
    for q in v.iter() { h = h ^ (*q as u64); }
    println("{}", h);
}
