// (0.30.1, RT-CG-3) tests/incremental.sh builds this with one cache as
// each variant after the other: `//` (one loop over v), then nothing (a
// second loop: std's slice iterator's `next` is called twice in the
// program), and the other way round. 0.30.0's warm build kept the first
// build's code of `next`, made while it had one call.

    let v: Vec<u64, Gpa> = (1u64..9u64).collect_in(gpa);
    let mut h: u64 = 1;
    for q in v.iter() { h = h ^ *q; }
    println("{}", h);
    @V@let mut k: u64 = 3; for q in v.iter() { k = k ^ *q; } println("{}", k);
}
