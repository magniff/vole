// VARIANTS: ga gb
// (0.30.1, RT-TL-2) An extern static's #[link_name] changed between two
// builds with one cache: user reads the new symbol (0.30.0's warm -O1 and
// -O2 builds read the old one).
extern "C" {
    #[link_name = "@V@"]
    static G: i32;
}

#[inline(never)]
fn user() -> i32 {
    // UNSAFE: a C static. SAFETY: read only, never written.
    unsafe { G }
}

fn main() {
    println("{}", user());
}
